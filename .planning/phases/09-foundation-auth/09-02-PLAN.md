Function signature:
```javascript
app.showLogin = function() {
  const appDiv = document.getElementById('app');
  appDiv.innerHTML = /* form HTML */;
  
  const form = document.getElementById('loginForm');
  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    // Validate, call API, store token
    const response = await app.apiCall('POST', '/api/auth/login', {
      email: form.email.value,
      password: form.password.value
    });
    
    if (response.ok) {
      const data = await response.json();
      localStorage.setItem('token', data.token);
      app.token = data.token;
      app.user = data.user;
      window.location.hash = '#/';
    } else {
      // Show error
    }
  });
};
```

Note: Token validation on app init happens in Plan 01 (app.js). This plan focuses on login form and localStorage.

```javascript
async loadPage(path) {
  const appDiv = document.getElementById('app');
  appDiv.innerHTML = '<div class="spinner"></div>';
  
  try {
    if (path === '/' || path === '') {
      this.showHome();
    } else if (path === 'login') {
      this.showLogin();
    } else if (path === 'register') {
      this.showRegister();
    } else if (path === 'reset-password') {
      this.showResetPassword();
    } else if (path.startsWith('reset-password-confirm')) {
      const token = new URLSearchParams(window.location.search).get('token');
      this.showResetPasswordConfirm(token);
    } else if (path === 'logout') {
      this.handleLogout();
    } else {
      appDiv.innerHTML = '<p>Page not found</p>';
    }
  } catch (e) {
    console.error('Error loading page:', e);
    appDiv.innerHTML = '<div class="alert alert-error">Error loading page</div>';
  }
}
```
