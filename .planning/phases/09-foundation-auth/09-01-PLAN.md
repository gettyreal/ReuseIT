// Client-side router for SPA
const app = {
  currentPage: null,
  token: localStorage.getItem('token'),
  user: null,
  
  // Initialize the app
  async init() {
    console.log('App initialized');
    
    // Check if user is logged in
    if (this.token) {
      try {
        // Verify token is still valid
        const response = await fetch('http://localhost:8000/api/auth/validate', {
          headers: { 'Authorization': `Bearer ${this.token}` }
        });
        if (!response.ok) {
          this.logout();
          return;
        }
        this.user = await response.json();
      } catch (e) {
        console.error('Token validation failed:', e);
        this.logout();
      }
    }
    
    // Set up route listeners
    this.setupRouter();
    
    // Navigate to initial page
    const path = window.location.hash.slice(2) || '/';
    this.navigate(path);
  },
  
  // Simple router
  setupRouter() {
    window.addEventListener('hashchange', () => {
      const path = window.location.hash.slice(2) || '/';
      this.navigate(path);
    });
  },
  
  // Navigate to page
  navigate(path) {
    // Protected routes check
    const protectedRoutes = ['profile', 'my-listings', 'bookings', 'chat', 'favorites'];
    if (protectedRoutes.some(r => path.startsWith(r)) && !this.token) {
      window.location.hash = '#/login';
      return;
    }
    
    console.log('Navigating to:', path);
    this.currentPage = path;
    this.loadPage(path);
  },
  
  // Load page content
  async loadPage(path) {
    const appDiv = document.getElementById('app');
    
    // Clear previous content
    appDiv.innerHTML = '<div class="spinner"></div>';
    
    try {
      if (path === '/' || path === '') {
        this.showHome();
      } else if (path === 'login') {
        this.showLogin();
      } else if (path === 'register') {
        this.showRegister();
      } else if (path === 'logout') {
        this.handleLogout();
      } else {
        appDiv.innerHTML = '<p>Page not found</p>';
      }
    } catch (e) {
      console.error('Error loading page:', e);
      appDiv.innerHTML = '<p>Error loading page</p>';
    }
  },
  
  // Page renderers (placeholder - actual pages in Plan 02)
  showHome() {
    const appDiv = document.getElementById('app');
    appDiv.innerHTML = '<h2>Home</h2><p>Welcome to ReuseIT</p>';
  },
  
  showLogin() {
    const appDiv = document.getElementById('app');
    appDiv.innerHTML = '<h2>Login</h2><p>Login form will be here</p>';
  },
  
  showRegister() {
    const appDiv = document.getElementById('app');
    appDiv.innerHTML = '<h2>Register</h2><p>Register form will be here</p>';
  },
  
  // API helper
  async apiCall(method, endpoint, body = null) {
    const headers = { 'Content-Type': 'application/json' };
    if (this.token) {
      headers['Authorization'] = `Bearer ${this.token}`;
    }
    
    const options = { method, headers };
    if (body) options.body = JSON.stringify(body);
    
    return fetch(`http://localhost:8000${endpoint}`, options);
  },
  
  // Logout
  logout() {
    localStorage.removeItem('token');
    this.token = null;
    this.user = null;
    window.location.hash = '#/login';
  },
  
  // Handle logout action
  async handleLogout() {
    try {
      await this.apiCall('POST', '/api/auth/logout');
    } catch (e) {
      console.error('Logout error:', e);
    }
    this.logout();
  }
};

// Start app when DOM is ready
document.addEventListener('DOMContentLoaded', () => app.init());
