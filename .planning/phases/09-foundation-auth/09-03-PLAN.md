```javascript
logout() {
  // Clear localStorage
  localStorage.removeItem('token');
  localStorage.removeItem('user');
  
  // Clear app state
  this.token = null;
  this.user = null;
  this.currentPage = null;
  
  // Redirect to login
  window.location.hash = '#/login';
  
  console.log('Logged out successfully');
},

// Handle logout action (called when user visits #/logout)
async handleLogout() {
  try {
    // Call backend logout endpoint to clear server-side session
    const response = await this.apiCall('POST', '/api/auth/logout');
    
    if (!response.ok) {
      console.warn('Server logout failed, clearing client-side session anyway');
    }
  } catch (e) {
    console.error('Error calling logout endpoint:', e);
    // Continue with client-side logout even if API call fails
  }
  
  // Clear client-side session
  this.logout();
}
```

**Key behavior:**
- logout() clears localStorage, app state, and redirects to login
- handleLogout() calls backend /api/auth/logout endpoint first, then clears client-side
- If backend logout fails, client-side logout still proceeds (fail-safe)
- User sees immediate redirect to login page

Ensure these functions replace the stubs created in Plan 01.

  <action>
Update the app.init() function in public/js/app.js to validate the token on startup:

```javascript
async init() {
  console.log('App initializing...');
  
  // Read token from localStorage
  this.token = localStorage.getItem('token');
  
  // If user is logged in, validate token with backend
  if (this.token) {
    try {
      const response = await this.apiCall('GET', '/api/auth/validate');
      
      if (response.ok) {
        // Token is valid, load user data
        const data = await response.json();
        this.user = data.user;
        console.log('Token validated, user logged in:', this.user.email);
      } else {
        // Token is invalid or expired
        console.warn('Token validation failed:', response.status);
        this.logout(); // Force logout
        return;
      }
    } catch (e) {
      console.error('Token validation error:', e);
      // On network error, trust the token (user might be offline)
      // Only logout if we get a clear 401/403 response
    }
  } else {
    console.log('No token found, user is not logged in');
  }
  
  // Set up router
  this.setupRouter();
  
  // Navigate to initial page
  const path = window.location.hash.slice(2) || '/';
  this.navigate(path);
}
```

**Key behavior:**
- On every page load, app checks if token is valid
- If token missing or invalid, forces logout and redirects to #/login
- If token valid, loads user data and allows page access
- Network errors don't trigger logout (offline scenarios)
- Token validation happens before router setup

**Note:** The /api/auth/validate endpoint should exist on backend (v1.0). If it doesn't, adjust the endpoint call or create a stub that always returns 200 if token header is present.

```javascript
navigate(path) {
  // Define which routes require authentication
  const protectedRoutes = [
    'profile',
    'edit-profile',
    'my-listings',
    'create-listing',
    'edit-listing',
    'bookings',
    'chat',
    'favorites',
    'reviews'
  ];
  
  // Check if this route requires authentication
  const isProtected = protectedRoutes.some(route => path.startsWith(route));
  
  // If route is protected and user is not logged in, redirect to login
  if (isProtected && !this.token) {
    console.warn('Attempted to access protected route without auth:', path);
    window.location.hash = '#/login';
    return;
  }
  
  console.log('Navigating to:', path);
  this.currentPage = path;
  this.loadPage(path);
}
```

**Key behavior:**
- Before loading any protected page, checks if user has token
- If no token, redirects to #/login
- Public routes (/, /login, /register, /reset-password, /logout, /search, /listing/:id) don't require auth
- User can browse listings without logging in, but can't create bookings or favorite items

**Protected vs Public Routes:**

```
Protected (require login):

- /reviews — submit review

Public (no auth required):
- / — home/browse listings
- /login — login page
- /register — register page
- /reset-password — password reset
- /logout — logout action
- /search — search results
- /listing/:id — view listing detail
```

```javascript
setupSessionTimeout() {
  // Auto-logout after 30 minutes of inactivity
  const SESSION_TIMEOUT = 30 * 60 * 1000; // 30 minutes in ms
  let timeoutId;
  
  const resetTimeout = () => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => {
      if (this.token) {
        console.warn('Session timeout due to inactivity');
        alert('Your session has expired. Please log in again.');
        this.logout();

    }, SESSION_TIMEOUT);
  };
  
  // Reset timeout on any user activity
  ['click', 'keypress', 'scroll'].forEach(event => {
    document.addEventListener(event, resetTimeout, true);
  });

}
```
